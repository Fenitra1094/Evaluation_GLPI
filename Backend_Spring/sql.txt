-- Table parent
CREATE TABLE categories (
  id   INTEGER PRIMARY KEY AUTOINCREMENT,
  nom  TEXT NOT NULL
);

-- Table enfant avec relation
CREATE TABLE produits (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  nom          TEXT NOT NULL,
  categorie_id INTEGER,
  FOREIGN KEY (categorie_id) REFERENCES categories(id)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);
